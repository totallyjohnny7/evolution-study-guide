--- JS FUNCTIONS ADDED (after resetCardTransform, before dkRender) ---

function cancelAutoTimer(){
  clearTimeout(_autoTimer);clearTimeout(_autoFlipTimer);_autoTimer=null;_autoFlipTimer=null;
  if(_autoCountIv){clearInterval(_autoCountIv);_autoCountIv=null;}
  var fill=$('dk-timer-fill');if(fill){fill.style.transition='none';fill.style.width='100%';}
  var cnt=$('dk-timer-cnt');if(cnt)cnt.textContent='';
}
function startAutoTimer(){
  cancelAutoTimer();if(!_deck.length)return;
  var fill=$('dk-timer-fill');var cnt=$('dk-timer-cnt');
  if(fill){fill.style.transition='none';fill.style.width='100%';fill.offsetWidth;fill.style.transition='width 10s linear';fill.style.width='0';}
  var secs=10;if(cnt)cnt.textContent=secs;
  _autoCountIv=setInterval(function(){secs--;if(secs>0){if(cnt)cnt.textContent=secs;}else{clearInterval(_autoCountIv);_autoCountIv=null;if(cnt)cnt.textContent='';}},1000);
  _autoTimer=setTimeout(function(){
    if(_autoCountIv){clearInterval(_autoCountIv);_autoCountIv=null;}if(cnt)cnt.textContent='';
    if(_animating||!$('dk-ov').classList.contains('show'))return;
    $('dk-card').classList.add('flipped');
    _autoFlipTimer=setTimeout(function(){if($('dk-ov').classList.contains('show'))flashGrade(false);},1500);
  },10000);
}

--- HOW IT WORKS ---

1. Every time a new card is shown (dkRender called), the 10-second timer starts.
2. A gold bar below the progress bar shrinks from 100% → 0% over 10 seconds.
3. A countdown number (10, 9, 8...) appears next to the card counter.
4. If the user does ANYTHING (touch, swipe, flip, grade, nav, shuffle, close) the timer
   cancels instantly.
5. If the timer reaches 0: card auto-flips to show the answer, then after 1.5 seconds
   automatically grades as MISS and advances to the next card.
